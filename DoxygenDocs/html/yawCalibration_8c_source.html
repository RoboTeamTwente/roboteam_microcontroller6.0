49void yaw_Calibrate(float newXsensYaw, float visionYaw, bool visionAvailable, float rateOfTurn) {
50 static float yawOffset = 0.0f;
51 static int restCounter = 0;
52 static float sumXsensVec[2] = {0.0f};
53 static float sumVisionVec[2] = {0.0f};
54
55 // Calibrate the xsens yaw from some time ago with vision to account for delay while sending.
56 float oldXsensYaw = getOldXsensYaw(newXsensYaw);
57
58 if (isCalibrationNeeded(visionYaw, oldXsensYaw, yawOffset) && visionAvailable && isRotatingSlow(rateOfTurn)) {
59 // TODO replace TIME_DIFF with hardware clock. If someone changes the timer, it would break everything
60 if (restCounter > CALIBRATION_TIME / TIME_DIFF) {
61 // calculate offset
62 float avgVisionYaw = atan2f(sumVisionVec[1], sumVisionVec[0]);
63 float avgXsensYaw = atan2f(sumXsensVec[1], sumXsensVec[0]);
64 yawOffset = constrainAngle(avgVisionYaw - avgXsensYaw);
65 hasCalibratedOnce = true;
66 restCounter = 0;
67 } else {
68 // Sum the unit vectors with these angles and then take the angle of the resulting vector.
69 sumXsensVec[0] += cosf(oldXsensYaw);
70 sumXsensVec[1] += sinf(oldXsensYaw);
71 sumVisionVec[0] += cosf(visionYaw);
72 sumVisionVec[1] += sinf(visionYaw);
73 restCounter++;
74 }
75 } else {
76 restCounter = 0;
77 sumXsensVec[0] = 0.0f;
78 sumXsensVec[1] = 0.0f;
79 sumVisionVec[0] = 0.0f;
80 sumVisionVec[1] = 0.0f;
81 }
82
83 // Add the calculated offset to the newly measured xsens yaw.
84 calibratedYaw = constrainAngle(newXsensYaw + yawOffset);
85
86}

106static bool isCalibrationNeeded(float visionYaw, float xsensYaw, float yawOffset) {
107 static bool calibrationNeeded = false;
108 static int checkCounter = 0;
109 if (!hasCalibratedOnce){
110 return true;
111 }
112
113 if (fabs(constrainAngle(visionYaw - (xsensYaw + yawOffset))) > M_PI/180/4) { // require 0.25 degree accuracy
114 checkCounter++;
115 } else {
116 checkCounter = 0;
117 calibrationNeeded = false;
118 }
119 if (checkCounter > 10) {
120 checkCounter = 0;
121 calibrationNeeded = true;
122 }
123 return calibrationNeeded;
124}
