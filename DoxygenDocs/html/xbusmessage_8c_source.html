77size_t XbusMessage_format(uint8_t* raw, struct XbusMessage const* message, enum XbusLowLevelFormat format)
78{
79 uint8_t* dptr = raw;
80 switch (format)
81 {
82 case XLLF_I2c:
83 {
84 *dptr++ = XBUS_CONTROL_PIPE;
85 }
86 break;
87
88 case XLLF_Spi:
89 {
90 *dptr++ = XBUS_CONTROL_PIPE;
91 // Fill bytes required to allow MT to process data
92 *dptr++ = 0;
93 *dptr++ = 0;
94 *dptr++ = 0;
95 }
96 break;
97
98 case XLLF_Uart:
99 {
100 *dptr++ = XBUS_PREAMBLE;
101 *dptr++ = XBUS_MASTERDEVICE;
102 }
103 break;
104 }
105
106 uint8_t checksum = (uint8_t)(-XBUS_MASTERDEVICE);
107
108 *dptr = message->mid;
109 checksum -= *dptr++;
110
111 uint16_t length = messageLength(message);
112
113 if (length < XBUS_EXTENDED_LENGTH)
114 {
115 *dptr = length;
116 checksum -= *dptr++;
117 }
118 else
119 {
120 *dptr = XBUS_EXTENDED_LENGTH;
121 checksum -= *dptr++;
122 *dptr = length >> 8;
123 checksum -= *dptr++;
124 *dptr = length & 0xFF;
125 checksum -= *dptr++;
126 }
127
128 formatPayload(dptr, message);
129 for (int i = 0; i < length; ++i)
130 {
131 checksum -= *dptr++;
132 }
133 *dptr++ = checksum;
134
135 return dptr - raw;
136}

146static uint8_t const* getPointerToData(enum XsDataIdentifier id, uint8_t const* data, uint16_t dataLength)
147{
148 uint8_t const* dptr = data;
149 while (dptr < data + dataLength)
150 {
151 uint16_t itemId;
152 uint8_t itemSize;
153 dptr = XbusUtility_readU16(&itemId, dptr);
154 dptr = XbusUtility_readU8(&itemSize, dptr);
155
156 if (id == itemId)
157 return dptr;
158
159 dptr += itemSize;
160 }
161 return NULL;
162}
