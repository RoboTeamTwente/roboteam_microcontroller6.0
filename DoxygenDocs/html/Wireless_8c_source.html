97// init functions
98Wireless_Error Wireless_Init(Wireless* w, SX1280_Settings set, SX1280_Interface* interface){
99 // Check inputs
100 if (w==NULL || interface==NULL){
101 return WIRELESS_PARAM_ERROR;
102 }
103 // fill Wireless struct with initial values
104 w->Settings = set;
105 w->Interface = interface;
106 w->Interface->active_transfer = false;
107 w->state = WIRELESS_INIT;
108 w->TXSyncword = 0x0;
109 w->RXSyncwords[0] = 0x0;
110 w->RXSyncwords[1] = 0x0;
111 w->readbufdest = NULL;
112 w->readbufBytes = 0;
113 w->TXchannel = WIRELESS_CHANNEL_DEFAULT_ROBOT_TO_BASESTATION;
114 w->RXchannel = WIRELESS_CHANNEL_DEFAULT_BASESTATION_TO_ROBOT;
115
116 // Start SX1280
117
118 /* 14.3 FLRC Operation, page 121 */
119 SX1280WakeUp(w->Interface); // reset, initialize SPI, set to STDBY_RC in order to configure SX1280
120
121 /* 6.1 Overview, page 34. Note: Care must therefore be taken to ensure that modulation parameters are set
122 using the command SetModulationParam() only after defining the packet type SetPacketType() to be used. */
123 setPacketType(w->Interface, w->Settings.packettype); // packet type is set first!
124 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
125
126 setChannel(w->Interface, w->RXchannel); // calls setRFFrequency() with freq=(channel+2400)*1000000
127 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
128
129 setBufferBase(interface, w->Settings.TXoffset, w->Settings.RXoffset); // set offsets to write TX and RX data in the SX1280 buffer
130 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
131
132 setModulationParam(w->Interface, &w->Settings.ModParam); // flrc = bitrate&BW, coding rate, modulation
133 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
134
135 setPacketParam(w->Interface, &w->Settings.PacketParam); // flrc = preamble length, syncword length, syncword match, header, payload length, crc length, whitening=no
136 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
137
138 setSyncSensitivity(w->Interface, w->Settings.syncSensitivity); // enable/disable highest gain step for receive
139 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
140
141 setSyncWordTolerance(w->Interface, w->Settings.syncWordTolerance); // set how many bits can go wrong in the sync word
142 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
143
144 setCrcSeed(w->Interface, w->Settings.crcSeed); // set crc seed value
145 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
146
147 setCrcPoly(w->Interface, w->Settings.crcPoly); // set crc polynomial
148 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
149
150 setTXParam(w->Interface, w->Settings.TXpower, w->Settings.TXrampTime); // set TX power and ramp up time
151 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
152
153 setDIOIRQParams(w->Interface, w->Settings.DIOIRQ); // IRQ masks are stored in reverse, so reverse again
154 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
155
156 // AutoFS reduces time between consecutive read / write operations
157 setAutoFS(w->Interface, true); // to go or not to go FS after TX or RX
158 if(validateCommand(w) == WIRELESS_ERROR) return WIRELESS_ERROR;
159
160 w->state = WIRELESS_READY;
161 return WIRELESS_OK;
162};

178Wireless_Error Wireless_setChannel(Wireless* w, WIRELESS_CHANNEL channel){
179 switch (channel){
180 case YELLOW_CHANNEL:
181 w->TXchannel = WIRELESS_CHANNEL_YELLOW_ROBOT_TO_BASESTATION;
182 w->RXchannel = WIRELESS_CHANNEL_YELLOW_BASESTATION_TO_ROBOT;
183 break;
184 case BLUE_CHANNEL:
185 w->TXchannel = WIRELESS_CHANNEL_BLUE_ROBOT_TO_BASESTATION;
186 w->RXchannel = WIRELESS_CHANNEL_BLUE_BASESTATION_TO_ROBOT;
187 break;
188 default:
189 w->TXchannel = WIRELESS_CHANNEL_YELLOW_ROBOT_TO_BASESTATION;
190 w->RXchannel = WIRELESS_CHANNEL_YELLOW_BASESTATION_TO_ROBOT;
191 break;
192 }
193 // If the robot was in receiving mode, switch to the new frequency and continue receiving
194 if(w->state == WIRELESS_RECEIVING){
195 setChannel(w->Interface, w->RXchannel);
196 if(w->continuousreceive){
197 WaitForPacketContinuous(w);
198 }else{
199 WaitForPacket(w);
200 }
201 }
202 return WIRELESS_OK;
203}

221Wireless_Error Wireless_setRXSyncwords(Wireless* w, uint32_t syncwords[2]){
222 // Determine which syncword channels should be used
223 bool useSync0 = syncwords[0] != 0;
224 bool useSync1 = syncwords[1] != 0;
225
226 // look-up new syncword mask
227 SX1280_MatchSyncWord newSyncwordMask = RX_DISABLE_SYNC_WORD;
228 if(useSync0 && useSync1){
229 newSyncwordMask = RX_MATCH_SYNC_WORD_2_3;
230 } else if(useSync0 && !useSync1){
231 newSyncwordMask = RX_MATCH_SYNC_WORD_2;
232 } else if(!useSync0 && useSync1){
233 newSyncwordMask = RX_MATCH_SYNC_WORD_3;
234 }
235
236 // Set new syncwords, only to register write if needed
237 if(useSync0 && w->RXSyncwords[0] != syncwords[0]){
238 w->RXSyncwords[0] = syncwords[0];
239 if(setSyncWord(w->Interface, 2, syncwords[0]) != SX1280_OK){
240 return WIRELESS_ERROR;
241 }
242 }
243 if(useSync1 && w->RXSyncwords[1] != syncwords[1]){
244 w->RXSyncwords[1] = syncwords[1];
245 if(setSyncWord(w->Interface, 3, syncwords[1]) != SX1280_OK){
246 return WIRELESS_ERROR;
247 }
248 }
249 // Update packetparam if the newSyncwordMask is different from the current matchsyncword
250 if(newSyncwordMask != w->Settings.PacketParam.matchsyncword){
251 w->Settings.PacketParam.matchsyncword = newSyncwordMask;
252 setPacketParam(w->Interface, &w->Settings.PacketParam);
253 }
254 return WIRELESS_OK;
255}

257// SX1280 buffer interface functions
258Wireless_Error WritePacket(Wireless* w, Wireless_Packet* packet){
259 if(w->state != WIRELESS_READY)
260 return WIRELESS_ERROR;
261
262 w->state = WIRELESS_WRITING;
263
264 clearIRQ(w->Interface,IRQ_ALL);
265 // If the packet that we're sending has a different size than the previous packet, we have to update the packet size in the SX1280
266 // Table 14-38: Payload Length Definition in FLRC Packet, page 124
267 if(w->Settings.PacketParam.payloadsize != packet->payloadLength){
268 w->Settings.PacketParam.payloadsize = packet->payloadLength;
269 setPacketParam(w->Interface, &w->Settings.PacketParam);
270 }
271 writeBuffer(w->Interface, w->Settings.TXoffset, packet->message, packet->payloadLength);
272 w->state = WIRELESS_READY;
273 return WIRELESS_OK;
274}

294Wireless_Error ReadPacket(Wireless* w, Wireless_Packet* packet){
295 w->state = WIRELESS_READING;
296
297 clearIRQ(w->Interface,IRQ_ALL);
298 SX1280_RX_Buffer_Status bs;
299 getRXBufferStatus(w->Interface, &bs);
300 readBuffer(w->Interface, bs.bufferOffset, packet->message, bs.payloadLength);
301 packet->payloadLength = bs.payloadLength;
302
303 w->state = w->continuousreceive ? WIRELESS_RECEIVING : WIRELESS_READY;
304 return WIRELESS_OK;

349Wireless_Error Wireless_IRQ_Handler(Wireless* w){
350 uint16_t irq;
351 SX1280_Packet_Status ps;
352 getIRQ(w->Interface, &irq);
353
354 clearIRQ(w->Interface,IRQ_ALL);
355
356 bool callback_handled = false;
357
358 if(irq & IRQ_CRC_ERROR) {
359 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_CRC_ERROR\n");
360 if(w->irqcallbacks && w->irqcallbacks->crcerror){
361 w->irqcallbacks->crcerror();
362 callback_handled = true;
363 }
364 }
365
366 /* Packet sent to robot */
367 if(irq & IRQ_TX_DONE){
368 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_TX_DONE\n");
369 if(w->state == WIRELESS_TRANSMITTING) w->state = WIRELESS_READY;
370 getPacketStatus(w->Interface, &ps);
371 if(w->irqcallbacks && w->irqcallbacks->txdone){
372 w->irqcallbacks->txdone(&ps);
373 callback_handled = true;
374 }
375 }
376
377 // Note : IRQ_RX_DONE also triggers when there is an crc error, so ignore IRQ_RX_DONE when IRQ_CRC_ERROR is also set: 16.3 All Modems: Interrupt with Bad CRC, page 150
378 if(irq & IRQ_RX_DONE && !(irq & IRQ_CRC_ERROR)){
379 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_RX_DONE\n");
380 getPacketStatus(w->Interface, &ps);
381 if(w->irqcallbacks && w->irqcallbacks->rxdone){
382 w->irqcallbacks->rxdone(&ps);
383 callback_handled = true;
384 }
385 }
386
387 if(irq & IRQ_RXTX_TIMEOUT) {
388 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_RXTX_TIMEOUT\n");
389 if(w->irqcallbacks && w->irqcallbacks->rxtxtimeout){
390 w->irqcallbacks->rxtxtimeout();
391 callback_handled = true;
392 }
393 }
394
395 if(irq & IRQ_SYNCWORD_VALID) {
396 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_SYNCWORD_VALID\n");
397 if(w->irqcallbacks && w->irqcallbacks->syncvalid){
398 w->irqcallbacks->syncvalid();
399 callback_handled = true;
400 }
401 }
402
403 if(irq & IRQ_SYNCWORD_ERROR) {
404 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_SYNCWORD_ERROR\n");
405 if(w->irqcallbacks && w->irqcallbacks->syncerror){
406 w->irqcallbacks->syncerror();
407 callback_handled = true;
408 }
409 }
410
411 if(irq & IRQ_PREAMBLE_DETECTED) {
412 if(w->printf) w->printf("[Wireless_IRQ_Handler] IRQ_PREAMBLE_DETECTED\n");
413 if(w->irqcallbacks && w->irqcallbacks->preambledetected){
414 w->irqcallbacks->preambledetected();
415 callback_handled = true;
416 }
417 }
418
419 if(!callback_handled){
420 if(w->printf) w->printf("[Wireless_IRQ_Handler] Default IRQ Handler for %d\n", irq);
421 if(w->irqcallbacks && w->irqcallbacks->default_callback){
422 w->irqcallbacks->default_callback();
423 }else{
424 /* WARNING! An IRQ has not been handled properly. This should never happen */
425 if(w->printf){
426 w->printf("[Wireless.c][Wireless_IRQ_Handler] Warning : interrupt not handled!\n");
427 }else{
428 /* Unsure what to do here? How to let people know that an interrupt has not been handled? */
429 }
430 }
431 }
432
433 return WIRELESS_OK;
434};

436Wireless_Error Wireless_DMA_Handler(Wireless* w){
437 switch (w->state){
438 case WIRELESS_READING:
439 DMA_Callback(w->Interface, w->readbufdest, w->readbufBytes);
440 if(w->continuousreceive){
441 w->state = WIRELESS_RECEIVING;
442 } else{
443 w->state = WIRELESS_READY;
444 }
445 if(w->rdcallback){
446 w->rdcallback();
447 w->rdcallback = NULL;
448 }
449 return WIRELESS_OK;
450
451 case WIRELESS_WRITING:
452 DMA_Callback(w->Interface, NULL, 0);
453 w->state = WIRELESS_READY;
454 if(w->wrcallback){
455 w->wrcallback();
456 w->wrcallback = NULL;
457 }
458 return WIRELESS_OK;
459
460 default:
461 if(w->printf){
462 w->printf("Wireless DMA handler called while in the wrong state");
463 }
464 return WIRELESS_ERROR;
465 }
466}

470Wireless_Error validateCommand(Wireless* w){
471 // Get the status of the SX after the last transaction to see if the command was processed correctly
472 SX1280_CommandStatus stat = getStatus(w->Interface).CommandStatus;
473 // C
474 switch (stat)
475 {
476 case Success:
477 return WIRELESS_OK;
478 case DataAvailable:
479 return WIRELESS_OK;
480 case TimeOut:
481 if(w->printf){
482 w->printf("ERROR: SX command TimeOut");
483 }
484 return WIRELESS_ERROR;
485 case ProcessingErr:
486 if(w->printf){
487 w->printf("ERROR: SX command ProcessingErr");
488 }
489 return WIRELESS_ERROR;
490 case Failed:
491 if(w->printf){
492 w->printf("ERROR: SX command Failed");
493 }
494 return WIRELESS_ERROR;
495 case TxDone:
496 return WIRELESS_OK;
497 default:
498 break;
499 }
500 return WIRELESS_OK;
501}
