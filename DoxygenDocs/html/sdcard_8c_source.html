26bool SDCard_Init(){
27 // Don't initialize if already initialized
28 if(sd_initialized) return true;
29
30 FRESULT result;
31 DIR directory;
32 FILINFO file_info;
33
34 // Mount the SD card immediately
35 result = f_mount(&fs, "", 1);
36 if(result != FR_OK) return false;
37
38 // Open the root directory
39 result = f_opendir(&directory, "");
40 if(result != FR_OK) return false;
41
42 // List all files on the sdcard, and count the number of log files, by checking if the filename ends with ".BIN"
43 // Also immediately delete any empty log files
44 uint32_t n_log_files = 0;
45
46 for(;;){
47 result = f_readdir(&directory, &file_info); // Read a directory item
48 if(result != FR_OK || file_info.fname[0] == 0) break; // Break on error or end of directory
49 if(file_info.fname[0] == '.') continue; // Ignore root directory
50 // Delete empty files while we're at it.
51 if(file_info.fsize == 0) {
52 f_unlink(file_info.fname); // Delete the file
53 // LOG_printf("[sdcard.c:"STRINGIZE(__LINE__)"] Deleted empty file: %s\n", file_info.fname);
54 }else{
55 if(strstr(file_info.fname, ".BIN") != NULL) n_log_files++; // If the filename ends with ".BIN", increment the counter
56 }
57 }
58
59 // Create a new log file name
60 char logFileName[20];
61 sprintf(logFileName, "LOG%04d.BIN", n_log_files);
62 LOG_printf("[sdcard.c:"STRINGIZE(__LINE__)"] Creating new log file: %s\n", logFileName);
63
64 // Open the log file, and overwrite it if it already exists (which it shouldn't)
65 result = f_open(&file, logFileName, FA_CREATE_ALWAYS | FA_WRITE);
66 if(result != FR_OK) return false;
67
68 // Set the time of the last sync to now
69 time_last_sync = HAL_GetTick();
70
71 sd_initialized = true;
72 return true;
73}

75bool SDCard_Write(uint8_t* data, uint32_t length, bool no_sync){
76 if(!sd_initialized) return false;
77
78 uint32_t bytes_written = 0;
79 FRESULT result = f_write(&file, data, length, &bytes_written);
80
81 // Increment the total number of bytes written
82 total_bytes_written += bytes_written;
83
84 // If the number of bytes written is not equal to the number of bytes we wanted to write,
85 // return false. This indicates that the SD card is full.
86 if(length != bytes_written) return false;
87 if(result != FR_OK) return false;
88
89 // If more than 30 seconds passed, sync the file to the SD card. This ensures that the data
90 // is written to the SD card. However, we don't want to do this too often, because it takes
91 // a lot of time, and it can cause the robot to stutter. Measurements indicated that it can
92 // take up to 10ms to sync the file.
93 if(!no_sync && HAL_GetTick() - time_last_sync > 15000){
94 __HAL_TIM_SET_COUNTER(TIM_1us, 0);
95 result = f_sync(&file);
96 uint32_t sync_time = __HAL_TIM_GET_COUNTER(TIM_1us);
97 // LOG_printf("[sdcard.c:"STRINGIZE(__LINE__)"] sync time: %luus\n", sync_time);
98 if(result != FR_OK) return false;
99 time_last_sync = HAL_GetTick();
100 }
101
102 return true;
103}
