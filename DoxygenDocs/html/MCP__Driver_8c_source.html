29void MCP_Init(CAN_HandleTypeDef *hcan, uint8_t board_id){
30 sending_board_id = board_id;
31
32 // Configuration of CAN filter
33 CAN_FilterTypeDef canfilterconfig;
34 canfilterconfig.FilterActivation = CAN_FILTER_ENABLE;
35 canfilterconfig.FilterMode = CAN_FILTERMODE_IDMASK;
36 canfilterconfig.FilterScale = CAN_FILTERSCALE_32BIT;
37 canfilterconfig.FilterBank = 10;
38 canfilterconfig.FilterFIFOAssignment = CAN_RX_FIFO0;
39 canfilterconfig.SlaveStartFilterBank = 0;
40 uint32_t total_filter = (MCP_LOCAL_VERSION << MCP_VERSION_BIT_SHIFT) | (sending_board_id << MCP_TO_ID_BIT_SHIFT);
41 uint32_t total_mask = MCP_VERSION_BIT_MASK | MCP_TO_ID_BIT_MASK;
42 // MSB bits 28:13
43 canfilterconfig.FilterIdHigh = (total_filter & 0x1FFFE000) >> 13;
44 canfilterconfig.FilterMaskIdHigh = (total_mask & 0x1FFFE000) >> 13;
45 // LSB bits 12:0
46 canfilterconfig.FilterIdLow = (total_filter & 0x1FFF) << 3;
47 canfilterconfig.FilterMaskIdLow = (total_mask & 0x1FFF) << 3;
48
49 // Mailboxes
50 MailBox_one.empty = true;
51 MailBox_one.message_id = 0;
52 MailBox_two.empty = true;
53 MailBox_two.message_id = 0;
54 MailBox_three.empty = true;
55 MailBox_three.message_id = 0;
56
57 // Acknowledgements
58 for (int i = 0; i < MCP_MAX_ID_PLUS_ONE; i++) {
59 ack_numbers[i] = 0x00;
60 free_to_send[i] = true;
61 }
62
63 // Configure CAN filter
64 HAL_CAN_ConfigFilter(hcan, &canfilterconfig);
65
66 // Start CAN communication
67 HAL_CAN_Start(hcan);
68 HAL_CAN_ActivateNotification(hcan, CAN_IT_RX_FIFO0_MSG_PENDING);
69}

103void MCP_Send_Message(CAN_HandleTypeDef *hcan, uint8_t *payload, CAN_TxHeaderTypeDef CAN_TxHeader, uint8_t to_board) {
104 if ((CAN_TxHeader.ExtId & MCP_ERROR_BIT_MASK) >> MCP_ERROR_BIT_SHIFT == 1) MCP_error_LOG(&CAN_TxHeader);

173bool extract_command(uint8_t RxData[], CAN_RxHeaderTypeDef *Header){
174 uint32_t message_ID = Header->ExtId;
175 uint8_t data[8];
176 memset(data, 0, sizeof(data));
177
178 // Copy received data to local array
179 for (int i = 0; i < Header->DLC; i++)
180 data[i] = RxData[i];
181
182 if (MCP_ID_IS_TYPE_ACK(message_ID)) {
183 uint8_t from_board = (message_ID & MCP_FROM_ID_BIT_MASK) >> MCP_FROM_ID_BIT_SHIFT;
184 uint8_t received_number = data[0];
185 if (ack_numbers[from_board] == received_number) {
186 ack_numbers[from_board] = (ack_numbers[from_board] + 1) % 0xFF;
187 free_to_send[from_board] = true;
188 MCP_resetSendMsg();
189 }
190
191 return false;
192 }
193
194 if (MCP_ID_IS_TYPE_ALIVE(message_ID)) {
195 uint8_t from_board = (message_ID & MCP_FROM_ID_BIT_MASK) >> MCP_FROM_ID_BIT_SHIFT;
196 free_to_send[from_board] = true;
197 }
198
199 // Check and store the command in the appropriate mailbox
200 if(MailBox_one.empty){
201 MailBox_one.empty = false;
202 MailBox_one.message_id = message_ID;
203 memcpy(MailBox_one.data_Frame, data, sizeof(data));
204 } else if (MailBox_two.empty){
205 MailBox_two.empty = false;
206 MailBox_two.message_id = message_ID;
207 memcpy(MailBox_two.data_Frame, data, sizeof(data));
208 } else if (MailBox_three.empty){
209 MailBox_three.empty = false;
210 MailBox_three.message_id = message_ID;
211 memcpy(MailBox_three.data_Frame, data, sizeof(data));
212 }
213 return true;
214}
