8def CamelCaseToUpper(word):
9 # Find all indices where a lowercase letter is followed by an uppercase letter
10 indices = [ i for i in range(1, len(word)) if word[i-1].islower() and word[i].isupper() ]
11 # Insert a _ at all of those indices
12 for i in reversed(indices): word = word[:i] + "_" + word[i:]
13 # Uppercase word
14 word = word.upper()
15
16 return word
