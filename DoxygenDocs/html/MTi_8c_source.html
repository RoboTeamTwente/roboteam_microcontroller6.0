24// Storage for messages to and from the MTi, must be alligned for possible DMA transfers
25
26// Cache coherency rules for RXbuf and TXbuf:
27// 1. Invalidate DCache if a new message is received. This forces the program to read from memory instead of the DCache (SCB_InvalidateDCache_by_Addr)
28// 2. Clean the DCache after a write action to TXbuf. This forces the program to write the DCache to memory. (SCB_CleanDCache_by_Addr)
29// 3. The SCB functions require that the data it is pointing to is 32 BYTE alligned, and as such are invalidated/cleaned with 32 BYTE multiples,
30// so the buffer alignement and size should match that.
31
32// This define makes sure the buffer length is a multiple of 32
33#define BYTES_NEEDED(BYTES) ((((BYTES - 1) >> 5) + 1) << 5)
34
35uint8_t RXbuf[BYTES_NEEDED(MAX_RAW_MESSAGE_SIZE)] __attribute__((aligned(32)));
36uint8_t TXbuf[BYTES_NEEDED(MAX_RAW_MESSAGE_SIZE)] __attribute__((aligned(32)));

77MTi_data* MTi_Init(uint16_t calibrate_time, enum XsFilterProfile filter_type) {
78 // init state for UART
79 MTi = &MTi_struct;
80 MTi->SPI = MTi_SPI;
81 MTi->CS_pin = MTi_NSS_pin;
82 MTi->SPI_busy = false;
83 MTi->init_phase = true;
84 MTi->RX_state = Xsens_receive_5;
85 MTi->RxDataFlag = 0;
86 MTi->ReceivedMessageStorage = malloc(MAX_RAW_MESSAGE_SIZE); // Reserve memory to store the longest possible message
87 MTi->TxBuffer = TXbuf;
88 MTi->RxBuffer = RXbuf;
89
90 // Create a structure to contain the callback functions
91 struct XbusParserCallback XBP_callback = {
92 .handleMessage = XBP_handleMessage,
93 .allocateBuffer = XBP_allocateBuffer,
94 .deallocateBuffer = XBP_deallocateBuffer
95 };
96 MTi->XBParser = XbusParser_create(&XBP_callback);
97 if (MTi->XBParser == NULL) {
98 LOG("[XSens:"STRINGIZE(__LINE__)"] Failed to create XBParser\n");
99 return NULL;
100 }
101
102 // start MTi operation of specific sensors and their update frequencies
103 set_Pin(MTi_RST_pin, false);
104 set_Pin(MTi->CS_pin, true);
105 HAL_Delay(1);
106 set_Pin(MTi_RST_pin, true);
107 //HAL_Delay(1);
108
109 if (WaitForAck(MTi, XMID_WakeUp)) {
110 LOG("[XSens:"STRINGIZE(__LINE__)"] Failed WakeUp\n");
111 return NULL;
112 }
113 uint8_t txbuf[6];
114 uint8_t rxbuf[6];
115
116 // set settings
117 HAL_Delay(1);
118 txbuf[0] = SetProtocol;
119 txbuf[HEADER_LENGTH] = Notification;//| Measurement;
120 while (MTi->SPI_busy) {}
121 MTi->SPI_busy = true;
122 set_Pin(MTi->CS_pin, false);
123 HAL_SPI_TransmitReceive(MTi->SPI, txbuf, rxbuf, 5, 100);
124 set_Pin(MTi->CS_pin, true);
125 MTi->SPI_busy = false;
126
127 // verify settings
128 HAL_Delay(1);
129 txbuf[0] = ReadProtocol;
130 while (MTi->SPI_busy) {}
131 MTi->SPI_busy = true;
132 set_Pin(MTi->CS_pin, false);
133 HAL_SPI_TransmitReceive(MTi->SPI, txbuf, rxbuf, 6, 100);
134 set_Pin(MTi->CS_pin, true);
135 MTi->SPI_busy = false;
136 if (MT_DEBUG) LOG_printf("[XSens:"STRINGIZE(__LINE__)"] Settings: %u\n", rxbuf[5]);
137
138 HAL_Delay(1);
139 MTi_GoToConfig(MTi);
140 if (WaitForAck(MTi, XMID_GoToConfigAck)) {
141 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed GoToConfig()\n");
142 return NULL;
143 }
144 // set measurement configurations
145 HAL_Delay(1);
146 MTi->data_configurations = data_configurations;
147 MTi->configuration_total = sizeof(data_configurations) / sizeof(MTi_data_tuple);
148
149 for (int i = 0; i < MTi->configuration_total; i++) {
150 MTi_data_tuple conf = data_configurations[i];
151 bool last = (i == (MTi->configuration_total - 1));
152 MTi_BuildConfig(MTi, conf.ID, conf.frequency, last);
153 }
154
155 if (WaitForAck(MTi, XMID_ReqOutputConfigurationAck)) {
156 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed ReqOutputConfigurationAck");
157 }
158
159 // set filter profile defined in XsFilterProfile
160 // 0 = general, 1 = high magnetic dep, 2 = dynamic
161 // 3 = assumes stable magnetic field, 4 = high dynamic magnetic field (unreferenced heading)
162 HAL_Delay(1);
163 if (MTi_SetFilterProfile(MTi, filter_type)) {
164 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed SetFilterProfile()");
165 return NULL;
166 }
167 // set MTi to measure state
168 HAL_Delay(1);
169 if (MTi_GoToMeasure(MTi)) {
170 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed GoToMeasure()");
171 return NULL;
172 }
173 MTi->init_phase = false;
174 HAL_Delay(100);
175 // calibrate rotation sensor for calibrate_time seconds
176 if (MTi_NoRotation(MTi, calibrate_time)) {
177 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed NoRotation()");
178 return NULL;
179 }
180
181 HAL_Delay(1);
182 if (MTi_UseIcc(MTi) == Xsens_OK) {
183 MTi->started_icc = true;
184 }
185 else {
186 LOG("[Xsens:"STRINGIZE(__LINE__)"] Failed UseIcc()");
187 return NULL;
188 }
189
190 /* Uncommented this because it's double? */
191 /* DONT UNCOMMENT THIS! IT DOESN'T WORK WITHOUT IT */
192 HAL_Delay(1);
193 if (MTi_UseIcc(MTi) == Xsens_OK) {
194 MTi->started_icc = true;
195 LOG_printf("ICC started");
196 }
197 else {
198 LOG_printf("ICC failed");
199 return NULL;
200 }
201
202 // Experiment to try and enable AHS
203 // Conclusion for now: it does not seem to work
204 // HAL_Delay(1);
205 // if (MTi_EnableAHS(MTi) == Xsens_OK) {
206 // LOG_printf("AHS enabled");
207 // }
208 // else {
209 // LOG_printf("AHS failed");
210 // }
211
212 // set settings
213 HAL_Delay(1);
214 txbuf[0] = SetProtocol;
215 txbuf[HEADER_LENGTH] = Notification | Measurement;
216 while (MTi->SPI_busy) {}
217 set_Pin(MTi->CS_pin, false);
218 HAL_SPI_TransmitReceive(MTi->SPI, txbuf, rxbuf, 5, 100);
219 set_Pin(MTi->CS_pin, true);
220
221 MTi->acc[0] = 0.0f;
222 MTi->acc[1] = 0.0f;
223 MTi->acc[2] = 0.0f;
224 return MTi;
225}

236Xsens_Status MTi_SPI_RxCpltCallback(MTi_data* MTi) {
237 // deselect MTi
238 set_Pin(MTi->CS_pin, true);
239 // Invalidate (entire) Cached RxBuffer to force load from memory
240 SCB_InvalidateDCache_by_Addr((uint32_t*)MTi->RxBuffer, BYTES_NEEDED(MAX_RAW_MESSAGE_SIZE));
241 MTi->RxBuffer[2] = XBUS_PREAMBLE;
242 MTi->RxBuffer[3] = XBUS_MASTERDEVICE;
243 XbusParser_parseBuffer(MTi->XBParser, MTi->RxBuffer + 2, MTi->RxBuffer[5] + 5);
244 MTi->SPI_busy = false;
245 Decode_Packet(MTi);
246 TheAlligator(MTi->XBParser);
247 return Xsens_OK;
248}

254Xsens_Status MTi_IRQ_Handler(MTi_data* MTi) {
255 Xsens_Status ret = Xsens_OK;
256
257 while (MTi->SPI_busy) {}
258 MTi->SPI_busy = true;
259 set_Pin(MTi->CS_pin, false);
260 uint8_t txbuf[4 + 4] = { 0x04,0x0 };
261 uint8_t rxbuf[4 + 4] = { 0x0 };
262 HAL_SPI_TransmitReceive(MTi->SPI, txbuf, rxbuf, 8, 100);
263 set_Pin(MTi->CS_pin, true);
264 MTi->SPI_busy = false;
265
266 //XbusParser_parseBuffer(MTi->XBParser, MTi->aRxBuffer, 5);
267 PipeStat* pipe_status = (PipeStat*)&rxbuf[HEADER_LENGTH];
268 if (pipe_status->notification_size) {
269 while (MTi->SPI_busy) {}
270 MTi->SPI_busy = true;
271
272 uint32_t message_size = 2 + pipe_status->notification_size;
273
274 uint8_t* ptr = MTi->TxBuffer;
275 *ptr++ = ReadNotification;
276 memset(ptr, 0, message_size);
277 // HAL_Delay(1); // MTi cannot process commands this fast, so add a delay
278 // Force (entire) MTi->TxBuffer write to memory
279
280 SCB_CleanDCache_by_Addr((uint32_t*)MTi->TxBuffer, BYTES_NEEDED(message_size + 2));
281 // Perform read/write action
282 set_Pin(MTi->CS_pin, false);
283 HAL_SPI_TransmitReceive(MTi->SPI, MTi->TxBuffer, MTi->RxBuffer, message_size + 2, 100);
284 set_Pin(MTi->CS_pin, true);
285
286 MTi->RxBuffer[2] = XBUS_PREAMBLE;
287 MTi->RxBuffer[3] = XBUS_MASTERDEVICE;
288 XbusParser_parseBuffer(MTi->XBParser, MTi->RxBuffer + 2, message_size);
289 MTi->SPI_busy = false;
290
291 if (MTi->ReceivedMessageStorage->mid == XMID_Error) {
292 ErrorHandler(MTi->ReceivedMessageStorage);
293 MTi->MT_Data_succerr[1]++; //failed
294 ret = Xsens_Failed_Update;
295 }
296 else {
297 processAck(MTi);
298 MTi->MT_Data_succerr[0]++; // success
299 }
300 //TheAlligator(MTi->XBParser);
301 }
302 if (pipe_status->measurement_size && !MTi->init_phase) {
303
304
305 //HAL_Delay(1);
306 while (MTi->SPI_busy) {}
307 MTi->SPI_busy = true;
308
309 uint32_t message_size = 2 + pipe_status->measurement_size;
310
311 uint8_t* ptr = MTi->TxBuffer;
312 *ptr++ = ReadMeasurement;
313 memset(ptr, 0, message_size);
314 // Force (entire) MTi->TxBuffer write to memory
315 SCB_CleanDCache_by_Addr((uint32_t*)MTi->TxBuffer, BYTES_NEEDED(message_size + 2));
316 // read from
317 set_Pin(MTi->CS_pin, false);
318 if (HAL_SPI_TransmitReceive_DMA(MTi->SPI, MTi->TxBuffer, MTi->RxBuffer, message_size + 2) != HAL_OK) {
319 MTi->SPI_busy = false;
320 }
321 }
322 return ret;
323}

326static Xsens_Status MTi_BuildConfig(MTi_data* MTi, enum XsDataIdentifier XDI, uint16_t frequency, bool complete) {
327 static uint8_t n_configs = 0;
328 static struct OutputConfiguration config[MAX_XDI_CONFIGS];
329 config[n_configs].dtype = XDI;
330 config[n_configs++].freq = frequency;
331 if (complete) {
332 struct XbusMessage mess;
333 mess.mid = XMID_SetOutputConfiguration;
334 mess.length = n_configs;
335 mess.data = &config;
336 SendXbusMessage(MTi, mess);
337 n_configs = 0;
338 }
339 return Xsens_OK;
340}

355static Xsens_Status Decode_Packet(MTi_data* MTi) {
356 struct XbusMessage const* message = MTi->ReceivedMessageStorage;
357 if (!message) return Xsens_Failed_Decode;
358
359 // loop through all requested data/measurements
360 // NOTE: not every value has to be in every packet

368static HAL_StatusTypeDef SendXbusMessage(MTi_data* MTi, struct XbusMessage XbusMessage) {
369 size_t XbusMes_size = XbusMessage_format(MTi->TxBuffer, (struct XbusMessage const*)&XbusMessage, XLLF_Spi);
370 bool SPI_status = false;
371 uint32_t cnt = 0;
372 while (MTi->SPI_busy) {}
373 MTi->SPI_busy = true;
374 do {
375 // Force (entire) MTi->TxBuffer write to memory
376 SCB_CleanDCache_by_Addr((uint32_t*)MTi->TxBuffer, XbusMes_size);
377 // Perform the write
378 set_Pin(MTi->CS_pin, false);
379 SPI_status = HAL_SPI_Transmit/*_IT*/(MTi->SPI, MTi->TxBuffer, XbusMes_size, 100);
380 set_Pin(MTi->CS_pin, true);
381 cnt++;
382 } while (SPI_status && cnt < 20);
383 MTi->SPI_busy = false;
384 return SPI_status;
385}

439static Xsens_Status processAck(MTi_data* MTi) {
440 enum XsMessageId XMID = MTi->ReceivedMessageStorage->mid;
441 MTi->LastAck = XMID;
442 // process ack
443 switch (XMID) {
444 case XMID_SetNoRotationAck:
445 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] SetNoRotationAck\n");
446 break;
447 case XMID_ReqFilterProfileAck:
448 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] Req/Set FilterProfileAck\n"); // request and set share the same value
449 break;
450 case XMID_IccCommandAck:
451 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] IccCommandAck\n");
452 break;
453 case XMID_GoToConfigAck:
454 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] In config state\n");
455 MTi->Xstate = Xsens_Config;
456 break;
457 case XMID_GoToMeasurementAck:
458 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] In measurement state\n");
459 MTi->Xstate = Xsens_Measure;
460 break;
461 case XMID_WakeUp:
462 if (MT_DEBUG) LOG("[Xsens:"STRINGIZE(__LINE__)"] woke up\n");
463 // send Ack back to MTi to confirm
464 struct XbusMessage XbusMes = { XMID_WakeUpAck };
465 SendXbusMessage(MTi, XbusMes);
466 // go to config state
467 MTi->Xstate = Xsens_Config;
468 break;
469 case XMID_ReqOutputConfigurationAck:
470 PrintOutputConfig(MTi->ReceivedMessageStorage);
471 break;
472 default:
473 LOG_printf("[Xsens:"STRINGIZE(__LINE__)"] Unknown message type: %u", XMID);
474 break;
475 }
476 return Xsens_OK;
477}

528static inline Xsens_Status WaitForAck(MTi_data* MTi, enum XsMessageId XMID) {
529 bool timedout = false;
530 uint32_t cnt = 0;
531 while (MTi->LastAck != XMID && !timedout) {
532 HAL_Delay(10);
533 timedout = 200 < cnt++;
534 }
535 return timedout ? Xsens_Failed_Receive : Xsens_OK;
536}
