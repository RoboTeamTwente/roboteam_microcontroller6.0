98void MCP_Process_Message(mailbox_buffer *to_Process){
99 bool send_ack = true;
100
101 if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_ARE_YOU_ALIVE) {
102 MCP_Send_Im_Alive();
103 send_ack = false;
104 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_REBOOT) {
105 HAL_Delay(1000);
106 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_CHIP) {
107 MCP_ChipPayload* cp = (MCP_ChipPayload*) to_Process->data_Frame;
108 MCP_Chip mcp_chip = {0};
109 decodeMCP_Chip(&mcp_chip, cp);
110 shoot_Shoot(shoot_Chip,mcp_chip.shootPower);
111 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_KICK) {
112 MCP_KickPayload* kp = (MCP_KickPayload*) to_Process->data_Frame;
113 MCP_Kick mcp_kick = {0};
114 decodeMCP_Kick(&mcp_kick, kp);
115 shoot_Shoot(shoot_Kick,mcp_kick.shootPower);
116 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_KICKER_CHARGE) {
117 shoot_StartCharging();
118 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_KICKER_MCP_KICKER_STOP_CHARGE) {
119 shoot_DisableCharging();
120 }
121
122 if (send_ack) MCP_Send_Ack(&hcan, to_Process->data_Frame[0], to_Process->message_id);
123
124 to_Process->empty = true;
125 *to_Process->data_Frame = 0;
126 to_Process->message_id = 0;
127}
