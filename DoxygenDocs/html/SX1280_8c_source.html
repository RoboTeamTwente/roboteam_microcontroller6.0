10void SX1280WakeUp(SX1280_Interface* interface) {
11 // reset the device
12 set_Pin(interface->Reset, LOW);
13 HAL_Delay(1);
14 set_Pin(interface->Reset, HIGH);
15 // Wait for interface to finish setting up
16 while(read_Pin(interface->BusyPin)) {}
17
18 // claim active transfer as the previous active_transfer is not valid anymore after a reset
19 interface->active_transfer = true;
20
21 // send command
22 interface->TXbuf[0] = GET_STATUS;
23 set_Pin(interface->CS, LOW);
24 HAL_SPI_TransmitReceive(interface->SPI, interface->TXbuf, interface->RXbuf, 1, 100);
25 set_Pin(interface->CS, HIGH);
26
27 UNLOCK(interface)
28
29 while(read_Pin(interface->BusyPin)) {}
30 // TODO: Check if needed? After reset the default is standby
31 setStandby(interface, 0);
32}

357/* 11.9.1 SetDioIrqParams, page 96 */
358void setDIOIRQParams(SX1280_Interface* interface, uint16_t DIOIRQ[4]){
359 uint16_t tmp[4] ={0};
360 for(int i = 0; i<4; i++){
361 // reverse IRQ mask cuz it stores them flipped in memory (Big Endian <-> Little Endian)
362 tmp[i] = ((DIOIRQ[i] << 8) & 0xFF00) | ((DIOIRQ[i] >> 8) & 0x00FF);
363 }
364 uint8_t* ptr = interface->TXbuf;
365 // wait till send complete
366 LOCK(interface)
367 // make command
368 *ptr++ = SET_DIO_IRQ_PARAM;
369 memcpy(ptr,&tmp,8);
370 SendData(interface,9);
371 UNLOCK(interface)
372}

461// NOTE: Assumes callee has control of active_transfer
462SX1280_Error SendData(SX1280_Interface* interface, uint8_t Nbytes){
463 while(read_Pin(interface->BusyPin)) {}
464 SX1280_Error ret = SX1280_OK;
465
466 // wait till ready
467 while(interface->SPI->State != HAL_SPI_STATE_READY){}
468 // send/receive data
469 set_Pin(interface->CS, LOW);
470 HAL_StatusTypeDef stat = HAL_SPI_TransmitReceive(interface->SPI, interface->TXbuf, interface->RXbuf, Nbytes, 100);
471 set_Pin(interface->CS, HIGH);
472 if(stat == HAL_ERROR || stat == HAL_TIMEOUT){
473 ret = SX1280_FAIL;
474 }
475
476 // wait for interface to process command
477 while(read_Pin(interface->BusyPin)) {}
478 return ret;
479}

481// NOTE: Assumes callee has control of active_transfer
482SX1280_Error SendData_DMA(SX1280_Interface* interface, uint8_t Nbytes){

491 ret = SX1280_FAIL;
492 }
