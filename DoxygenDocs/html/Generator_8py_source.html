35 set1 = list(range(start, stop))

263 def to_packet_description(self, packet_name, variables:list[Variable]):
264 """ Generates a description of the packet, such as the packet name, the date, and a description of all variables
265
266 Args:
267 packet_name (str): The name of the packet
268 variables (list[Variable]): The variables in the packet
269
270 Returns:
271 str: The description of the packet
272 """
273 # Calculate number of bytes needed to store all variables. ceil(total number of bits / 8)
274 n_bytes = math.ceil(sum([n_bits for (_, n_bits, _, _) in variables]) / 8)
275 # Create [---0--] [---1--] [---2--] .......
276 string_description = self.begin_block_comment() + "\n"
277 string_description += ' '.join([ f"[{i : ^6}]" for i in range(n_bytes) ])
278 # Keep track of location in the packet
279 current_bit = 0
280 for variable, n_bits, _, desc in variables:
281 string_description += "\n"
282 active_bits = list(range(current_bit, current_bit + n_bits)) # Indices of active bits
283 for i in range(n_bytes * 8): # For all bits
284 if i != 0 and i % 8 == 0: string_description += " " # Add whitespace
285 string_description += "1" if i in active_bits else "-" # Add '1' if bit active, '-' if inactive
286 string_description += f" {variable}"
287 current_bit += n_bits
288 string_description += "\n" + self.end_block_comment() + "\n"
289 return string_description

370 def to_getter(self, packet_name:str, variable:Variable, current_bit:int) -> str:
371 """ Generates the body for the getter function, that gets a variable from a payload instance, decodes it, and returns it
372
373 Example for C REM_RobotCommand_get_rho:
374 uint32_t _rho = ((remrcp->payload[8]) << 8) | ((remrcp->payload[9]));
375 return (_rho * 0.0001220721751736F);
376
377 Args:
378 packet_name (str): The name of the packet
379 variable (Variable): The variable to create the getter for
380 current_bit (int): The current bit index within the encoded packet where the start of the variable is located
381
382 Returns:
383 str: The code for the body of the getter function
384 """
385
386 variable_name, n_bits, _range, desc = variable # Unpack Variable
387 at_byte, at_bit = current_bit // 8, current_bit % 8 # Calculate byte and bit index of current variable within the encoded packet
388 _type = getType(n_bits, _range) # Figure out the type of the variable
389 float_conversion = _range is not None # Check if the variable needs to be converted from integer to float
390
391 payload_variable = self.get_payload_variable(packet_name)
392
393 if _type == "bool":
394 return f"{self.get_indent()} return ({payload_variable}[{at_byte}] & {bitString(at_bit, at_bit+1)}) > 0;\n"
395
396 if _type in ["uint32_t", "uint64_t"]:
397 n_bits_remaining = n_bits
398 bitwise_operations = []
399
400 while 0 < n_bits_remaining:
401 # Find the byte index in the packet, and the bit index within that byte
402 at_byte, at_bit = current_bit // 8, current_bit % 8
403 # Calculate the number of bits to process from the current byte. Always 8 or less
404 # Either the remaining bits in the variable, or the remaining bits in the byte, whichever one is smaller
405 bits_from_byte = min(n_bits_remaining, 8 - at_bit)
406 # Figure out which bits we need from the byte
407 start_bit, stop_bit = at_bit, at_bit + bits_from_byte
408 # Generate bitshift mask, except if the entire byte is used, e.g. " & 0b00111000"
409 bitshift_mask = "" if bits_from_byte == 8 else f" & {bitString(start_bit, stop_bit)}"
410 # Figure out how much we need to shift this baby in total
411 shift_to_right = (8 - stop_bit)
412 shift_to_left = n_bits_remaining - bits_from_byte
413 # for C, figure out if type casting is needed
414 type_cast = self.to_typecast(shift_to_left)
415
416 bitwise_operation = f"({type_cast}({payload_variable}[{at_byte}]{bitshift_mask}){shift(shift_to_left, shift_to_right)})"
417 bitwise_operations.append(bitwise_operation)
418
419 n_bits_remaining -= bits_from_byte
420 current_bit += bits_from_byte
421
422 # Concatenate all bit operations
423 bitwise_operations_string = ' | '.join(bitwise_operations)
424
425 # No range given, so no conversion to float needed
426 if not float_conversion:
427 return f"{self.get_indent()} return {bitwise_operations_string};\n"
428 # Range given. Do integer => float conversion
429 else:
430 a, b = getConversion(n_bits, _range) # y = ax + b
431 variable_type = "" if self.convert_type(_type) is None else f"{self.convert_type(_type)} "
432 conversion_string = f"{self.get_indent()} {variable_type}_{variable_name} = {bitwise_operations_string};\n"
433 conversion_string += f"{self.get_indent()} return (_{variable_name} * {a:.16f}{self.to_float()})"
434 if b != 0: conversion_string += f" + {b:.16f}{self.to_float()}"
435 conversion_string += ";\n"
436 return conversion_string

438 def to_setter(self, packet_name:str, variable:Variable, current_bit:int) -> str:
439 """ Generates the body for the sets function, that encodes a value, and places it in a payload instance
440
441 Example for C REM_RobotCommand_set_rho:
442 uint32_t _rho = (uint32_t)(rho / 0.0001220721751736F);
443 remrcp->payload[8] = (_rho >> 8);
444 remrcp->payload[9] = _rho;
445
446 Args:
447 packet_name (str): The name of the packet
448 variable (Variable): The variable to create the getter for
449 current_bit (int): The current bit index within the encoded packet where the start of the variable is located
450
451 Returns:
452 str: The code for the body of the setter function
453 """
454 variable_name, n_bits, _range, desc = variable # Unpack Variable
455 at_byte, at_bit = current_bit // 8, current_bit % 8 # Calculate byte and bit index of current variable within the encoded packet
456 _type = getType(n_bits, _range) # Figure out the type of the variable
457 float_conversion = _range is not None # Check if the variable needs to be converted from integer to float
458
459 payload_variable = self.get_payload_variable(packet_name)
460
461 # Create the conversion from float to integer, if needed
462 conversion_string = ""
463 if float_conversion:
464 # Formula : y = ax + b => x = (y - b) / a
465 a, b = getConversion(n_bits, _range)
466 variable_type = "" if self.convert_type(_type) is None else f"{self.convert_type(_type)} "
467 subtract_string = f"{variable_name}" if b == 0 else f"({variable_name} {-b:+.16f}{self.to_float()})"
468 division_string = f" / {int(a)}" if int(a) == float(a) else f" / {a:.16f}{self.to_float()}"
469 calculation_string = self.cast_to_type(f"{subtract_string}{division_string}", _type)
470 conversion_string += f"{self.get_indent()} {variable_type}_{variable_name} = {calculation_string};\n"
471 variable_name = f"_{variable_name}"
472
473 # Create all bitwise operations
474 n_bits_remaining = n_bits
475 bitwise_operations = []
476
477 while 0 < n_bits_remaining:
478 # Find the byte index in the packet, and the bit index within that byte
479 at_byte, at_bit = current_bit // 8, current_bit % 8
480 # Calculate the number of bits to process from the current byte. Always 8 or less
481 # Either the remaining bits in the variable, or the remaining bits in the byte, whichever one is smaller
482 bits_from_byte = min(n_bits_remaining, 8 - at_bit)
483 # Figure out which bits we need from the byte
484 start_bit, stop_bit = at_bit, at_bit + bits_from_byte
485
486 # Generate bitshift mask, except if the entire byte is used, e.g. " & 0b00111000"
487 bitshift_mask = "" if bits_from_byte == 8 else f" & {bitString(start_bit, stop_bit)}"
488 # Generate inverse bitmask
489 bitmask_inverted = f" & {bitString(start_bit, stop_bit, True)}"
490
491 # Figure out how much we need to shift this baby in total
492 shift_to_left = 8 - stop_bit
493 shift_to_right = max(0, n_bits_remaining - bits_from_byte)
494
495 # Create the mask that prevents unrelated bits from being set
496 string_inverse = ""
497 if bits_from_byte < 8:
498 string_inverse = f" | ({payload_variable}[{at_byte}]{bitmask_inverted})"
499
500 # Add shifting of variable if needed
501 string_shift = variable_name
502 if shift_to_left != 0 or shift_to_right != 0:
503 string_shift = f"({variable_name}{shift(shift_to_left, shift_to_right)})"
504
505 # Add masking to variable if needed
506 string_bitmask = string_shift
507 if bits_from_byte != 8:
508 string_bitmask = f"({string_shift}{bitshift_mask})"
509
510 operation_string = f"{self.get_indent()} {payload_variable}[{at_byte}] = {string_bitmask}{string_inverse};"
511 bitwise_operations.append(operation_string)
512
513 n_bits_remaining -= bits_from_byte
514 current_bit += bits_from_byte
515
516 operations_string = conversion_string
517 operations_string += '\n'.join(bitwise_operations) + "\n"
518 return operations_string
