9def CamelCaseToUpper(word):
10 # Find all indices where a lowercase letter is followed by an uppercase letter
11 indices = [ i for i in range(1, len(word)) if word[i-1].islower() and word[i].isupper() ]
12 # Insert a _ at all of those indices
13 for i in reversed(indices): word = word[:i] + "_" + word[i:]
14 # Uppercase word
15 word = word.upper()
16
17 return word
