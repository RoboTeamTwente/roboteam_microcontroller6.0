370 def to_getter(self, packet_name:str, variable:Variable, current_bit:int) -> str:
371 """ Generates the body for the getter function, that gets a variable from a payload instance, decodes it, and returns it
372
373 Example for C REM_RobotCommand_get_rho:
374 uint32_t _rho = ((remrcp->payload[8]) << 8) | ((remrcp->payload[9]));
375 return (_rho * 0.0001220721751736F);
376
377 Args:
378 packet_name (str): The name of the packet
379 variable (Variable): The variable to create the getter for
380 current_bit (int): The current bit index within the encoded packet where the start of the variable is located
381
382 Returns:
383 str: The code for the body of the getter function
384 """
385
386 variable_name, n_bits, _range, desc = variable # Unpack Variable
387 at_byte, at_bit = current_bit // 8, current_bit % 8 # Calculate byte and bit index of current variable within the encoded packet
388 _type = getType(n_bits, _range) # Figure out the type of the variable
389 float_conversion = _range is not None # Check if the variable needs to be converted from integer to float
390
391 payload_variable = self.get_payload_variable(packet_name)
392
393 if _type == "bool":
394 return f"{self.get_indent()} return ({payload_variable}[{at_byte}] & {bitString(at_bit, at_bit+1)}) > 0;\n"
395
396 if _type in ["uint32_t", "uint64_t"]:
397 n_bits_remaining = n_bits
398 bitwise_operations = []
399
400 while 0 < n_bits_remaining:
401 # Find the byte index in the packet, and the bit index within that byte
402 at_byte, at_bit = current_bit // 8, current_bit % 8
403 # Calculate the number of bits to process from the current byte. Always 8 or less
404 # Either the remaining bits in the variable, or the remaining bits in the byte, whichever one is smaller
405 bits_from_byte = min(n_bits_remaining, 8 - at_bit)
406 # Figure out which bits we need from the byte
407 start_bit, stop_bit = at_bit, at_bit + bits_from_byte
408 # Generate bitshift mask, except if the entire byte is used, e.g. " & 0b00111000"
409 bitshift_mask = "" if bits_from_byte == 8 else f" & {bitString(start_bit, stop_bit)}"
410 # Figure out how much we need to shift this baby in total
411 shift_to_right = (8 - stop_bit)
412 shift_to_left = n_bits_remaining - bits_from_byte
413 # for C, figure out if type casting is needed
414 type_cast = self.to_typecast(shift_to_left)
415
416 bitwise_operation = f"({type_cast}({payload_variable}[{at_byte}]{bitshift_mask}){shift(shift_to_left, shift_to_right)})"
417 bitwise_operations.append(bitwise_operation)
418
419 n_bits_remaining -= bits_from_byte
420 current_bit += bits_from_byte
421
422 # Concatenate all bit operations
423 bitwise_operations_string = ' | '.join(bitwise_operations)
424
425 # No range given, so no conversion to float needed
426 if not float_conversion:
427 return f"{self.get_indent()} return {bitwise_operations_string};\n"
428 # Range given. Do integer => float conversion
429 else:
430 a, b = getConversion(n_bits, _range) # y = ax + b
431 variable_type = "" if self.convert_type(_type) is None else f"{self.convert_type(_type)} "
432 conversion_string = f"{self.get_indent()} {variable_type}_{variable_name} = {bitwise_operations_string};\n"
433 conversion_string += f"{self.get_indent()} return (_{variable_name} * {a:.16f}{self.to_float()})"
434 if b != 0: conversion_string += f" + {b:.16f}{self.to_float()}"
435 conversion_string += ";\n"
436 return conversion_string
