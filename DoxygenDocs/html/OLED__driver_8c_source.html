69void OLED_Update(button_id_t button, bool test_mode) {
70 /* return if not initialized */
71 if (!oled_initialized) {
72 return;
73 }
74
75 /* show error if a menu has too many children; afterwards return */
76 if (flag_error_too_many_children) {
77 if (!flag_error_too_many_children_page_init) menuHasTooManyChildrenException();
78 return;
79 }
80
81 /* if no button is pressed return; update variables displayed on page if needed beforehand */
82 if (button == BUTTON_NONE) {
83 if (current_page->is_test != NOT_A_TEST) {
84 run_test(current_page);
85 }
86 if (current_page->has_variables) {
87 clear_screen();
88 putPageName();
89 update_variables(current_page);
90 display_text();
91 SSD1306_UpdateScreen(); // update screen
92 }
93 return;
94 }
95
96 /* find out what to do on button press */
97 if (current_page->id == root_page.id || current_page->id == not_in_test_mode.id || current_page->id == error_menu_has_no_children.id) {
98 current_page = root_page.children[0];
99 item_selector = 0;
100 } else if (current_page->is_menu) {
101 onButtonPressMenu(button);
102 } else if (current_page->is_test != NOT_A_TEST) {
103 onButtonPressSelfTest(button);
104 } else {
105 onButtonPressDefault(button);
106 }
107
108 /* Prevent user from going into test menu if robot not in test mode*/
109 if (!test_mode && (current_page->id == id_self_test_menu || current_page->is_test != NOT_A_TEST)) {
110 current_page = &not_in_test_mode;
111 display_text();
112 }
113
114 /* Check if menu has items, otherwise throw error */
115 if (current_page->is_menu && current_page->n_of_children == 0) {
116 menuHasNoChildrenException();
117 } else {
118 refresh();
119 }
120 resetButtonState(button);
121}

228static void onButtonPressMenu(button_id_t button) {
229 int nomiator = current_page->n_of_children + 1;
230 switch(button){
231 case BUTTON_UP:
232 item_selector = (item_selector - 1 + nomiator) % nomiator;
233 break;
234 case BUTTON_DOWN:
235 item_selector = (item_selector + 1 + nomiator) % nomiator;
236 break;
237 case BUTTON_LEFT:
238 menu_move_sideways(-1);
239 item_selector = 0;
240 break;
241 case BUTTON_RIGHT:
242 menu_move_sideways(1);
243 item_selector = 0;
244 break;
245 case BUTTON_OK:
246 if (item_selector < current_page->n_of_children) {
247 current_page = current_page->children[item_selector];
248 } else {
249 current_page = current_page->parent;
250 }
251 item_selector = 0;
252 break;
253 default:
254 break;
255 }
256}

400static void menu_move_sideways(int direction) {
401 if (current_page->parent == NULL) {
402 return;
403 }
404
405 page_struct *parent = current_page->parent;
406 if (parent->n_of_children == 1) {
407 return;
408 }
409
410 //find index of current page
411 int index = 0;
412 for (int counter = 0; counter < parent->n_of_children; counter++) {
413 if (parent->children[counter]->id == current_page->id) {
414 index = counter;
415 break;
416 }
417 }
418
419 current_page = parent->children[(index + parent->n_of_children + direction) % parent->n_of_children];
420}
