186void XbusParser_parseByte(struct XbusParser* parser, const uint8_t byte)
187{
188 switch (parser->state)
189 {
190 case XBPS_Preamble:
191 if (byte == XBUS_PREAMBLE)
192 {
193 parser->checksum = 0;
194 parser->state = XBPS_BusId;
195 }
196 break;
197
198 case XBPS_BusId:
199 parser->checksum += byte;
200 parser->state = XBPS_MessageId;
201 break;
202
203 case XBPS_MessageId:
204 parser->checksum += byte;
205 parser->currentMessage.mid = (enum XsMessageId)byte;
206 parser->state = XBPS_Length;
207 break;
208
209 case XBPS_Length:
210 parser->checksum += byte;
211 if (byte == XBUS_NO_PAYLOAD)
212 {
213 parser->currentMessage.length = byte;
214 parser->currentMessage.data = NULL;
215 parser->state = XBPS_Checksum;
216 }
217 else if (byte < XBUS_EXTENDED_LENGTH)
218 {
219 parser->currentMessage.length = byte;
220 prepareForPayload(parser);
221 parser->state = XBPS_Payload;
222 }
223 else
224 {
225 parser->state = XBPS_ExtendedLengthMsb;
226 }
227 break;
228
229 case XBPS_ExtendedLengthMsb:
230 parser->checksum += byte;
231 parser->currentMessage.length = ((uint16_t)byte) << 8;
232 parser->state = XBPS_ExtendedLengthLsb;
233 break;
234
235 case XBPS_ExtendedLengthLsb:
236 parser->checksum += byte;
237 parser->currentMessage.length |= byte;
238 prepareForPayload(parser);
239 parser->state = XBPS_Payload;
240 break;
241
242 case XBPS_Payload:
243 parser->checksum += byte;
244 if (parser->currentMessage.data)
245 {
246 ((uint8_t*)parser->currentMessage.data)[parser->payloadReceived] = byte;
247 }
248 if (++parser->payloadReceived == parser->currentMessage.length)
249 {
250 parser->state = XBPS_Checksum;
251 }
252 break;
253
254 case XBPS_Checksum:
255 parser->checksum += byte;
256 if ((parser->checksum == 0) &&
257 ((parser->currentMessage.length == 0) ||
258 parser->currentMessage.data))
259 {
260 parseMessagePayload(parser);
261 parser->callbacks.handleMessage(&parser->currentMessage);
262 }
263 else if (parser->currentMessage.data)
264 {
265 parser->callbacks.deallocateBuffer(parser->currentMessage.data);
266 }
267 parser->state = XBPS_Preamble;
268 break;
269 }
270}
