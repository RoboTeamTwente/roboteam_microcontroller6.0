180 def to_getter(self, packet_name:str, variable:Variable, current_bit:int) -> str:
181 """ Generates the body for the getter function, that gets a variable from a payload instance, decodes it, and returns it
182
183 Example for C REM_RobotCommand_get_rho:
184 uint32_t _rho = ((remrcp->payload[8]) << 8) | ((remrcp->payload[9]));
185 return (_rho * 0.0001220721751736F);
186
187 Args:
188 packet_name (str): The name of the packet
189 variable (Variable): The variable to create the getter for
190 current_bit (int): The current bit index within the encoded packet where the start of the variable is located
191
192 Returns:
193 str: The code for the body of the getter function
194 """
195
196 variable_name, n_bits, _range, desc = variable # Unpack Variable
197 at_byte, at_bit = current_bit // 8, current_bit % 8 # Calculate byte and bit index of current variable within the encoded packet
198 _type = getType(n_bits, _range) # Figure out the type of the variable
199 float_conversion = _range is not None # Check if the variable needs to be converted from integer to float
200
201 payload_variable = self.get_payload_variable(packet_name)
202
203 if _type == "bool":
204 return f" return ({payload_variable}[{at_byte}] & {bitString(at_bit, at_bit+1)}) > 0;\n"
205
206 if _type in ["uint32_t", "uint64_t"]:
207 n_bits_remaining = n_bits
208 bitwise_operations = []
209
210 while 0 < n_bits_remaining:
211 # Find the byte index in the packet, and the bit index within that byte
212 at_byte, at_bit = current_bit // 8, current_bit % 8
213 # Calculate the number of bits to process from the current byte. Always 8 or less
214 # Either the remaining bits in the variable, or the remaining bits in the byte, whichever one is smaller
215 bits_from_byte = min(n_bits_remaining, 8 - at_bit)
216 # Figure out which bits we need from the byte
217 start_bit, stop_bit = at_bit, at_bit + bits_from_byte
218 # Generate bitshift mask, except if the entire byte is used, e.g. " & 0b00111000"
219 bitshift_mask = "" if bits_from_byte == 8 else f" & {bitString(start_bit, stop_bit)}"
220 # Figure out how much we need to shift this baby in total
221 shift_to_right = (8 - stop_bit)
222 shift_to_left = n_bits_remaining - bits_from_byte
223 # for C, figure out if type casting is needed
224 type_cast = self.to_typecast(shift_to_left)
225
226 bitwise_operation = f"({type_cast}({payload_variable}[{at_byte}]{bitshift_mask}){shift(shift_to_left, shift_to_right)})"
227 bitwise_operations.append(bitwise_operation)
228
229 n_bits_remaining -= bits_from_byte
230 current_bit += bits_from_byte
231
232 # Concatenate all bit operations
233 bitwise_operations_string = ' | '.join(bitwise_operations)
234
235 # No range given, so no conversion to float needed
236 if not float_conversion:
237 return f" return {bitwise_operations_string};\n"
238 # Range given. Do integer => float conversion
239 else:
240 a, b = getConversion(n_bits, _range) # y = ax + b
241 variable_type = "" if self.convert_type(_type) is None else f"{self.convert_type(_type)} "
242 conversion_string = f" {variable_type}_{variable_name} = {bitwise_operations_string};\n"
243 conversion_string += f" return (_{variable_name} * {a:.16f}F)"
244 if b != 0: conversion_string += f" + {b:.16f}F"
245 conversion_string += ";\n"
246 return conversion_string
