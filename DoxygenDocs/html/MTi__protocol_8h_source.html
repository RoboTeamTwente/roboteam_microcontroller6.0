20typedef enum MTi_opcodes {

23 SetPipe = 0x03, // (write) send messages (reduced Xbus message)

25 ReadNotification = 0x05, // (read) read notifications (reduced Xbus message)
26 ReadMeasurement = 0x06 // (read) read measurement data (reduced Xbus message)
27}MTi_opcodes;

35}drdyConf;
