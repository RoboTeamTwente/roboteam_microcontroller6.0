221void wheels_Update(ControlOutput* const output) {
222 /* Don't run the wheels if these are not initialized */
223 /* Not that anything would happen anyway, because the PWM timers wouldn't be running, but still .. */
224 if (!wheels_initialized) {
225 wheels_Stop();
226 return;
227 }
228
229 body2Wheels(wheelRef, stateLocalRef); //translate velocity to wheel speed
230 body2Wheels(wheelRefBodyScaled, stateLocalRefBodyScaled); //translate velocity to wheel speed
231
232 float vu = stateLocalRef[vel_u];
233 float vv = stateLocalRef[vel_v];
234 float rho = sqrt(vu * vu + vv * vv);
235 float theta_local = atan2(vv, vu);
236 float omega = stateLocalRef[vel_w];
237
238 float wheels_measured_speeds[4] = { 0.0f };
239 wheels_GetMeasuredSpeeds(wheels_measured_speeds);
240
241 float feed_forward[4] = { 0.0f };
242
243 for (motor_id_t motor = RF; motor <= RB; motor++) {
244
245 // Feedforward
246 double wheel_speed_threshold = 0.5f;
247 if (fabs(wheelRef[motor]) < wheel_speed_threshold) {
248 feed_forward[motor] = 0;
249 // Or should this be:
250 // feed_forward[motor] = feedforwardMass(stateLocalRefAcceleration,motor);
251 }
252 else {
253 feed_forward[motor] = wheelFFOn * (feedforwardParameters.identified_damping * wheelRef[motor] +
254 feedforwardFriction(wheelRef[motor], rho, theta_local, omega, motor)) +
255 massFFOn * feedforwardMass(stateLocalRefAcceleration, motor);
256 }
257
258 // Feedback
259 // Calculate the velocity error
260 float angular_velocity_error = wheelRefBodyScaled[motor] - wheels_measured_speeds[motor];
261
262 // If the error is very small, ignore it (why is this here?)
263 if (fabs(angular_velocity_error) < 0.1f) {
264 angular_velocity_error = 0.0f;
265 wheelsK[motor].I = 0.0f;
266 }
267
268 float feed_back_voltage = wheelFBOn * 24.0f * (0.001367311f * (PID(angular_velocity_error, &wheelsK[motor])));
269 float max_fb_voltage = 12.5f;
270 if (feed_back_voltage > max_fb_voltage) {
271 feed_back_voltage = max_fb_voltage;
272 }
273 else if (feed_back_voltage < -max_fb_voltage) {
274 feed_back_voltage = -max_fb_voltage;
275 }
276
277
278 // Set motor PWM fraction/voltage
279 float max_voltage = 12.5f;
280 // Add PID to commanded speed and convert to PWM (range between -1 and 1)
281 float wheel_voltage_to_be_applied = feed_forward[motor] + feed_back_voltage;
282 if (wheel_voltage_to_be_applied > max_voltage) {
283 wheel_voltage_to_be_applied = max_voltage;
284 }
285 else if (wheel_voltage_to_be_applied < -max_voltage) {
286 wheel_voltage_to_be_applied = -max_voltage;
287 }
288 wheel_speed_fraction[motor] = voltage2PWM(wheel_voltage_to_be_applied);
289 }
290 memcpy(&output->wheel_efforts, wheel_speed_fraction, 16);
291}

483static float feedforwardMass(float stateLocalRefAcceleration[3], motor_id_t motor) {
484 float bodyForce[4] = { 0.0f };
485 float wheelForceDivRw[4] = { 0.0f };
486 float wheelForce[4] = { 0.0f };
487 float wheelTorque[4] = { 0.0f };
488 float motorCurrent[4] = { 0.0f };
489 float motorVoltage[4] = { 0.0f };
490
491 bodyForce[0] = mass * stateLocalRefAcceleration[0];
492 bodyForce[1] = mass * stateLocalRefAcceleration[1];
493 bodyForce[2] = inertia * stateLocalRefAcceleration[2]; // 0.0 because inertia is set to 0 for now. So no angular acceleration feedforward
494 bodyForce[3] = 0.0f; // This is not used, it is just to have four inputs for body2wheels
495
496 body2Wheels(wheelForceDivRw, bodyForce); //translate velocity to wheel speed
497
498 wheelForce[motor] = wheelForceDivRw[motor] * rad_wheel;
499 wheelTorque[motor] = wheelForce[motor] * rad_wheel;
500 motorCurrent[motor] = wheelTorque[motor] / TORQUE_CONSTANT_MOTOR;
501 motorVoltage[motor] = motorCurrent[motor] * RESISTANCE_MOTOR;
502 return motorVoltage[motor];
503}

527static float detailedFitEval(float theta, float b, float c, float param1, float param2, float param3, float param4) {
528 float y = 0.0f;
529 float y_sign = sin(b * theta + c);
530 if (y_sign >= 0) {
531 y = param1 * sin(param2 * theta + param3) + param4;
532 }
533 else {
534 y = -(param1 * sin(param2 * theta + param3) + param4);

539static float feedforwardFriction(float wheelRef, float rho, float theta, float omega, wheel_names wheel) {
540
541 bool detailedFit = true;
542
543 // Calculations
544 float vw_max_round_to_rotational = feedforwardParameters.vw_max_round_to_rotational_scaling * (rho / rad_robot);
545 // Equal to 0.3
546 float z_rotational = feedforwardParameters.rotation_feedforward_value[wheel];
547
548 // Calculate the feedforward voltage needed for purely translating
549 float z_translation = 0.0f;
550 if (detailedFit) {
551 z_translation = detailedFitEval(theta, feedforwardParameters.b[wheel], feedforwardParameters.c[wheel], feedforwardParameters.param1[wheel], feedforwardParameters.param2[wheel], feedforwardParameters.param3[wheel], feedforwardParameters.param4[wheel]);
552 }
553 else {
554 z_translation = constsineEval(theta, feedforwardParameters.a[wheel], feedforwardParameters.b[wheel], feedforwardParameters.c[wheel], feedforwardParameters.d[wheel]);
555 }
556
557 // Make sure the feedforward voltage is always applied along the desired rotational direction of the wheel
558 int wheel_velocity_larger_than_zero;
559 if (wheelRef > 0) {
560 wheel_velocity_larger_than_zero = 1;
561 }
562 else {
563 wheel_velocity_larger_than_zero = 0;
564 }
565
566 float z_rotational_fixed;
567 float z_translation_fixed = z_translation;
568 if (wheel_velocity_larger_than_zero == 1) {
569 // Equal to 0.3
570 z_rotational_fixed = z_rotational;
571 if (z_translation < 0) {
572 if (detailedFit == true) {
573 z_translation_fixed = feedforwardParameters.param4[wheel];
574 }
575 else {
576 z_translation_fixed = feedforwardParameters.d[wheel];
577 }
578 }
579 }
580 else {
581 z_rotational_fixed = -1 * z_rotational;
582 if (z_translation > 0) {
583 if (detailedFit == true) {
584 z_translation_fixed = -feedforwardParameters.param4[wheel];
585 }
586 else {
587 z_translation_fixed = -feedforwardParameters.d[wheel];
588 }
589 }
590 }
591
592 // Define a smooth transition between translational and rotational feedforward gains.
593 float gamma = 0.5f - 0.5f * cos(M_PI * fabs(omega / vw_max_round_to_rotational));
594 if (fabs(omega) > vw_max_round_to_rotational) {
595 gamma = 1;
596 }
597 if (gamma > 1) {
598 gamma = 1;
599 }
600 float feedforwardFrictionValue = ((1 - gamma) * z_translation_fixed + gamma * z_rotational_fixed);
601 return feedforwardFrictionValue;
602}

604static void body2Wheels(float wheelSpeed[4], float stateLocal[4]) {
605
606 // Translate the local u, v, and omega velocities into wheel velocities.
607 wheelSpeed[wheels_RF] = stateLocal[vel_u] * D[0] + stateLocal[vel_v] * D[1] + stateLocal[vel_w] * D[2];
608 wheelSpeed[wheels_LF] = stateLocal[vel_u] * D[3] + stateLocal[vel_v] * D[4] + stateLocal[vel_w] * D[5];
609 wheelSpeed[wheels_LB] = stateLocal[vel_u] * D[6] + stateLocal[vel_v] * D[7] + stateLocal[vel_w] * D[8];
610 wheelSpeed[wheels_RB] = stateLocal[vel_u] * D[9] + stateLocal[vel_v] * D[10] + stateLocal[vel_w] * D[11];
611
612 // Translate wheel velocities into angular velocities
613 wheelSpeed[wheels_RF] = wheelSpeed[wheels_RF] / rad_wheel;
614 wheelSpeed[wheels_LF] = wheelSpeed[wheels_LF] / rad_wheel;
615 wheelSpeed[wheels_LB] = wheelSpeed[wheels_LB] / rad_wheel;
616 wheelSpeed[wheels_RB] = wheelSpeed[wheels_RB] / rad_wheel;
617
618}

620static void global2Local(float global[4], float local[4], float yaw_angle, float stateGlobalRefAcceleration[3], float stateLocalRefAcceleration[3]) {
621 //trigonometry
622 local[vel_u] = cosf(yaw_angle) * global[vel_x] + sinf(yaw_angle) * global[vel_y];
623 local[vel_v] = -sinf(yaw_angle) * global[vel_x] + cosf(yaw_angle) * global[vel_y];
624 local[vel_w] = global[vel_w];
625 local[yaw] = global[yaw];
626 stateLocalRefAcceleration[vel_u] = cosf(yaw_angle) * stateGlobalRefAcceleration[vel_x] + sinf(yaw_angle) * stateGlobalRefAcceleration[vel_y];
627 stateLocalRefAcceleration[vel_v] = -sinf(yaw_angle) * stateGlobalRefAcceleration[vel_x] + cosf(yaw_angle) * stateGlobalRefAcceleration[vel_y];
628 stateLocalRefAcceleration[vel_w] = stateGlobalRefAcceleration[vel_w];
629}

631static void velocityControl(float stateLocal[4], float stateGlobalRef[4], float velocityWheelRef[4]) {
632 global2Local(stateGlobalRef, stateLocalRef, stateLocal[yaw], stateGlobalRefAcceleration, stateLocalRefAcceleration); //transfer global to local
633
634 // Local control
635 float veluErr = (stateLocalRef[vel_u] - stateLocal[vel_u]);
636 float velvErr = (stateLocalRef[vel_v] - stateLocal[vel_v]);
