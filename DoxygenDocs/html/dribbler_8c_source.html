213int calculateDifference(int value1, int value2, int maxvalue) {
214 int directDiff = value2 - value1;
215 int wrapDiff;
216
217 // Calculate wrap-around difference
218 if (value1 > value2) {
219 wrapDiff = (maxvalue - value1 + value2 + 1);
220 } else {
221 wrapDiff = -(maxvalue - value2 + value1 + 1);
222 }
223
224 // Determine which difference is smaller in magnitude
225 if (abs(directDiff) < abs(wrapDiff)) {
226 return directDiff;
227 } else {
228 return wrapDiff;
229 }
230}
