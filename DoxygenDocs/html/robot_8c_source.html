317void updateTestCommand(REM_RobotCommand* rc, u32 time) {
318 // First, empty the entire RobotCommand
319 resetRobotCommand(rc);
320 // Set the basic required stuff
321 rc->packetType = REM_PACKET_TYPE_REM_ROBOT_COMMAND;
322 rc->remVersion = REM_LOCAL_VERSION;
323 rc->toRobotId = ROBOT_ID;
324
325 u32 c_time = time - get_system_test_time_started();
326
327 // Split up testing window into blocks of two seconds
328 float period_fraction = (c_time % 2000) / 2000.f;
329
330 // Rotate around, slowly
331 rc->angularVelocity = 6 * (float)sin(period_fraction * 2 * M_PI);
332 // Turn on dribbler
333 rc->dribblerOn = true;
334 // Kick a little every block

337 rc->kickChipPower = 1;
338 rc->doForce = true;

346void MCP_Process_Message(mailbox_buffer* to_Process) {
347 if (ROBOT_INITIALIZED) toggle_Pin(LED7_pin);
348 bool send_ack = true;
349 switch (to_Process->message_id) {
350 case MCP_PACKET_ID_POWER_TO_TOP_MCP_POWER_ALIVE:;
351 MCP_PowerAlivePayload* pap = (MCP_PowerAlivePayload*)to_Process->data_Frame;
352 decodeMCP_PowerAlive(&powerAlive, pap);
353 flag_PowerBoard_alive = true;
354 send_ack = false;
355 break;
356 case MCP_PACKET_ID_DRIBBLER_TO_TOP_MCP_DRIBBLER_ALIVE:;
357 MCP_DribblerAlivePayload* dap = (MCP_DribblerAlivePayload*)to_Process->data_Frame;
358 decodeMCP_DribblerAlive(&dribblerAlive, dap);
359 flag_DribblerBoard_alive = true;
360 send_ack = false;
361 break;
362 case MCP_PACKET_ID_KICKER_TO_TOP_MCP_KICKER_ALIVE:;
363 MCP_KickerAlivePayload* kap = (MCP_KickerAlivePayload*)to_Process->data_Frame;
364 decodeMCP_KickerAlive(&kickerAlive, kap);
365 flag_KickerBoard_alive = true;
366 send_ack = false;
367 if (ROBOT_INITIALIZED) {
368 MCP_KickerCharge kc = { 0 };
369 MCP_KickerChargePayload kcp = { 0 };
370 encodeMCP_KickerCharge(&kcp, &kc);
371 MCP_Send_Message(&hcan1, &kcp, kickerChargeHeader, MCP_KICKER_BOARD);
372 }
373 break;
374 case MCP_PACKET_ID_KICKER_TO_TOP_MCP_KICKER_CAPACITOR_VOLTAGE:;
375 MCP_KickerCapacitorVoltagePayload* kcvp = (MCP_KickerCapacitorVoltagePayload*)to_Process->data_Frame;
376 decodeMCP_KickerCapacitorVoltage(&kickerCapacitorVoltage, kcvp);
377 break;
378 case MCP_PACKET_ID_POWER_TO_TOP_MCP_POWER_VOLTAGE:;
379 MCP_PowerVoltagePayload* pvp = (MCP_PowerVoltagePayload*)to_Process->data_Frame;
380 decodeMCP_PowerVoltage(&powerVoltage, pvp);
381 break;
382 case MCP_PACKET_ID_DRIBBLER_TO_TOP_MCP_SEES_BALL:;
383 MCP_SeesBallPayload* spb = (MCP_SeesBallPayload*)to_Process->data_Frame;
384 decodeMCP_SeesBall(&seesBall, spb);
385 break;
386 case MCP_PACKET_ID_KICKER_TO_TOP_MCP_KICKER_STATUS:;
387 MCP_KickerStatusPayload* ksp = (MCP_KickerStatusPayload*)to_Process->data_Frame;
388 decodeMCP_KickerStatus(&kickerStatus, ksp);
389 break;
390 }
391
392 if (send_ack) MCP_Send_Ack(&hcan1, to_Process->data_Frame[0], to_Process->message_id);
393
394 LOG_sendAll();
395 to_Process->empty = true;
396 *to_Process->data_Frame = 0;
397 to_Process->message_id = 0;
398}

1017bool handlePacket(u8* packet_buffer, u8 packet_length) {
1018 u8 total_bytes_processed = 0;
1019 u8 packet_header;
1020
1021 while (total_bytes_processed < packet_length) {
1022
1023 packet_header = packet_buffer[total_bytes_processed];
1024
1025 switch (packet_header) {
1026
1027 case REM_PACKET_TYPE_REM_ROBOT_COMMAND:
1028 handleRobotCommand(packet_buffer + total_bytes_processed);
1029 total_bytes_processed += REM_PACKET_SIZE_REM_ROBOT_COMMAND;
1030 break;
1031 case REM_PACKET_TYPE_REM_ROBOT_BUZZER:
1032 handleRobotBuzzer(packet_buffer + total_bytes_processed);
1033 total_bytes_processed += REM_PACKET_SIZE_REM_ROBOT_BUZZER;
1034 break;
1035
1036 case REM_PACKET_TYPE_REM_ROBOT_GET_PIDGAINS:
1037 handleRobotGetPIDGains(packet_buffer + total_bytes_processed);
1038 total_bytes_processed += REM_PACKET_SIZE_REM_ROBOT_GET_PIDGAINS;
1039 break;
1040
1041 case REM_PACKET_TYPE_REM_ROBOT_SET_PIDGAINS:
1042 handleRobotSetPIDGains(packet_buffer + total_bytes_processed);
1043 total_bytes_processed += REM_PACKET_SIZE_REM_ROBOT_SET_PIDGAINS;
1044 break;
1045
1046 case REM_PACKET_TYPE_REM_ROBOT_MUSIC_COMMAND:
1047 handleRobotMusicCommand(packet_buffer + total_bytes_processed);
1048 total_bytes_processed += REM_PACKET_SIZE_REM_ROBOT_MUSIC_COMMAND;
1049 break;
1050
1051 case REM_PACKET_TYPE_REM_SX1280FILLER:
1052 total_bytes_processed += REM_PACKET_SIZE_REM_SX1280FILLER;
1053 break;
1054
1055 case REM_PACKET_TYPE_REM_ROBOT_KILL_COMMAND:
1056 handleRobotKillCommand(packet_buffer + total_bytes_processed);
1057 total_bytes_processed += REM_PACKET_TYPE_REM_ROBOT_KILL_COMMAND;
1058 break;
1059
1060
1061 default:
1062 LOG_printf("[REM Handler] Error! At %d of %d bytes. [Packet type] = %d\n", total_bytes_processed, packet_length, packet_header);
1063 return false;
1064 }
1065 }
1066
1067 return true;
1068}
