40int stateEstimation_Init() {
41 kalman_Init();
42
43 // Initialize the pseudoinverse of the velocity coupling matrix.
44 float divisor_a = cosFront * cosFront + cosBack * cosBack;
45 Dinv[0] = 0.5 * (cosFront / divisor_a);
46 Dinv[1] = -Dinv[0];
47 Dinv[2] = -0.5 * (cosBack / divisor_a);
48 Dinv[3] = -Dinv[2];
49
50 float divisor_b = sinBack + sinFront;
51 Dinv[4] = 0.5 * (1 / divisor_b);
52 Dinv[5] = Dinv[4];
53 Dinv[6] = -Dinv[4];
54 Dinv[7] = -Dinv[4];
55
56 Dinv[8] = 0.5 * (sinBack / divisor_b);
57 Dinv[9] = Dinv[8];
58 Dinv[10] = 0.5 * (sinFront / divisor_b);
59 Dinv[11] = Dinv[10];
60
61 return 0;
62}

114static void wheels2Body(float wheelSpeeds[4], float output[3]) {
115 float wheelSpeedsLinear[4]; // The linear speed for each wheel [m/s]
116
117 // Translate angular wheel velocities [rad/s] into regular velocities [m/s] on the outside of each wheel.
118 for (wheel_names wheel = wheels_RF; wheel <= wheels_RB; wheel++) {
119 wheelSpeedsLinear[wheel] = wheelSpeeds[wheel] * rad_wheel;
120 }
121
122 // Translate the wheel speeds into local u, v and r * w speeds.
123 output[vel_u] = wheelSpeedsLinear[0] * Dinv[0] + wheelSpeedsLinear[1] * Dinv[1] + wheelSpeedsLinear[2] * Dinv[2] + wheelSpeedsLinear[3] * Dinv[3];
124 output[vel_v] = wheelSpeedsLinear[0] * Dinv[4] + wheelSpeedsLinear[1] * Dinv[5] + wheelSpeedsLinear[2] * Dinv[6] + wheelSpeedsLinear[3] * Dinv[7];
125 output[vel_w] = wheelSpeedsLinear[0] * Dinv[8] + wheelSpeedsLinear[1] * Dinv[9] + wheelSpeedsLinear[2] * Dinv[10] + wheelSpeedsLinear[3] * Dinv[11];
126
127 // Translate the rotational velocity of the robot back to rad/s.
128 output[vel_w] = output[vel_w] / rad_robot;
129}

131float smoothen_rateOfTurn(float rateOfTurn) {
132 static float buffer[RoT_BUFFER_SIZE] = { 0.0f }; // circular buffer
133 static int idx = 0; // holds current index of buffer

137 idx = (idx + 1) % RoT_BUFFER_SIZE;

142 }
