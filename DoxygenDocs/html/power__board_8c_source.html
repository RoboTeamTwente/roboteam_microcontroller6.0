123/*
124 * The header is 13 bits
125 * The first two bits are for the IDE and RTR
126 * After that, the first 7 bits is the message ID
127 * While the left most 4 bits is used as the ID of the receiver
128 * The pay load will be a list of length 8 with 8 bit numbers, for a total of 64 bits or 8 bytes
129 */
130void MCP_Process_Message(mailbox_buffer *to_Process){
131
132 bool send_ack = true;
133
134 if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_POWER_MCP_ARE_YOU_ALIVE) {
135 MCP_Send_Im_Alive();
136 send_ack = false;
137 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_POWER_MCP_KILL) {
138 kill();
139 } else if (to_Process->message_id == MCP_PACKET_ID_TOP_TO_POWER_MCP_REBOOT) {
140 HAL_Delay(1000);
141 }
142
143 if (send_ack) MCP_Send_Ack(&hcan, to_Process->data_Frame[0], to_Process->message_id);
144
145 to_Process->empty = true;
146 *to_Process->data_Frame = 0;
147 to_Process->message_id = 0;
148}
